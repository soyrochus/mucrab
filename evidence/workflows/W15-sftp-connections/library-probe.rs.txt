use ssh2::{Session,HashType, RenameFlags};
use std::{net::{TcpStream,Shutdown},time::{Duration,Instant},path::Path,io::{Read,Write}};
fn main(){
 let a:Vec<_>=std::env::args().collect();let socket=TcpStream::connect(&a[1]).unwrap();socket.set_read_timeout(Some(Duration::from_secs(3))).unwrap();socket.set_write_timeout(Some(Duration::from_secs(3))).unwrap();let shutdown=socket.try_clone().unwrap();
 let mut session=Session::new().unwrap();session.set_tcp_stream(socket);session.set_timeout(3000);session.handshake().unwrap();assert!(!session.authenticated());println!("host key before auth: {:?}, sha256 bytes:{}",session.host_key().unwrap().1,session.host_key_hash(HashType::Sha256).unwrap().len());
 assert!(session.userauth_password(&a[2],"invalid-probe-only").is_err());session.userauth_pubkey_file(&a[2],None,Path::new(&a[3]),None).unwrap();assert!(session.authenticated());
 let sftp=session.sftp().unwrap();let root=Path::new(&a[4]);let old=root.join("old");let new=root.join("new");sftp.create(&old).unwrap().write_all(b"replacement").unwrap();sftp.create(&new).unwrap().write_all(b"original").unwrap();let replace=sftp.rename(&old,&new,Some(RenameFlags::ATOMIC|RenameFlags::OVERWRITE|RenameFlags::NATIVE));println!("generic rename overwrite result:{replace:?}");let mut bytes=vec![];sftp.open(&new).unwrap().read_to_end(&mut bytes).unwrap();println!("existing bytes:{}",String::from_utf8(bytes).unwrap());
 let mut channel=session.channel_session().unwrap();channel.subsystem("sftp").unwrap();channel.write_all(&[0,0,0,5,1,0,0,0,3]).unwrap();let mut length=[0;4];channel.read_exact(&mut length).unwrap();let mut version=vec![0;u32::from_be_bytes(length) as usize];channel.read_exact(&mut version).unwrap();assert_eq!(version[0],2);println!("protocol version:{} posix-rename advertised:{}",u32::from_be_bytes(version[1..5].try_into().unwrap()),version.windows(b"posix-rename@openssh.com".len()).any(|b|b==b"posix-rename@openssh.com"));
 let mut packet=vec![200];packet.extend_from_slice(&1u32.to_be_bytes());for s in ["posix-rename@openssh.com".to_owned(),old.to_str().unwrap().to_owned(),new.to_str().unwrap().to_owned()]{packet.extend_from_slice(&(s.len() as u32).to_be_bytes());packet.extend_from_slice(s.as_bytes());}channel.write_all(&(packet.len() as u32).to_be_bytes()).unwrap();channel.write_all(&packet).unwrap();channel.read_exact(&mut length).unwrap();let mut response=vec![0;u32::from_be_bytes(length) as usize];channel.read_exact(&mut response).unwrap();assert_eq!(response[0],101);assert_eq!(u32::from_be_bytes(response[5..9].try_into().unwrap()),0);let mut bytes=vec![];sftp.open(&new).unwrap().read_to_end(&mut bytes).unwrap();assert_eq!(bytes,b"replacement");println!("advertised posix-rename overwrite passed; prior destination never unlinked by client");
 let entry=keyring::Entry::new("org.mucrab.w15-probe","disposable").unwrap();println!("credential service available:{}",entry.get_password().is_ok());
 // Shutdown interrupts a pending channel read without waiting on the session mutex.
 let start=Instant::now();std::thread::spawn(move||{std::thread::sleep(Duration::from_millis(100));shutdown.shutdown(Shutdown::Both).unwrap();});let _=channel.read_exact(&mut length);println!("socket shutdown acknowledged in {}ms",start.elapsed().as_millis());
}
